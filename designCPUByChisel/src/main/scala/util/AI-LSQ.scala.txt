package nscscc.util

// 文件: cpu/mem/LSQ.scala
 
import chisel3._
import chisel3.util._
import nscscc.util.CircularQueuePtr
 
/**
 * Load/Store Queue 的指针定义
 *
 * 【关键需求】
 *   LSQ需要判断"一条load是否在一条store之后"，
 *   以检测存储器依赖违例（Store-to-Load Forwarding错误）
 *
 * 【isAfter 的威力】
 *   只需一条指令：  loadPtr.isAfter(storePtr)
 *   就能判断时序关系，不用自己算扩展地址
 */
class LqPtr(val entries: Int = 64) extends CircularQueuePtr[LqPtr](entries)
class SqPtr(val entries: Int = 64) extends CircularQueuePtr[SqPtr](entries)
 
/**
 * 演示：如何用 CircularQueuePtr 做存储器顺序检查
 */
class StoreToLoadCheck(lqSize: Int = 64, sqSize: Int = 64) extends Module {
  val io = IO(new Bundle {
    /** 一条load指令的队列指针 */
    val loadPtr  = Input(new LqPtr(lqSize))
    /** 一条store指令的队列指针 */
    val storePtr = Input(new SqPtr(sqSize))
    /** load是否在store之后（即store可能需要转发数据给load） */
    val loadIsAfterStore = Output(Bool())
  })
 
  // 注意：LqPtr 和 SqPtr 是不同的类型！
  // 它们属于不同的队列，不能直接比较。
  // 但如果您的LSQ是统一索引的，可以定义一个通用的 LsPtr。
 
  // 这里我们假设使用统一的指针来演示 isAfter：
  // io.loadIsAfterStore := io.loadPtr.isAfter(io.storePtr)
 
  // 实际使用中，如果load和store在同一个序空间：
  // （以下仅为演示，真实LSQ会更复杂）
  io.loadIsAfterStore := false.B  // 占位，实际设计中替换为真实逻辑
}

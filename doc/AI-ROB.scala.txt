package nscscc.util

// 文件: cpu/backend/ROB.scala
 
package cpu.backend
 
import chisel3._
import chisel3.util._
import nscscc.util.CircularQueuePtr
 
/**
 * 重排序缓冲区 (Reorder Buffer)
 *
 * 【在CPU中的位置】
 *   乱序执行引擎的核心组件，负责：
 *   ① 按序提交指令（乱序执行完成 → 按序写回）
 *   ② 管理指令的推测执行状态
 *   ③ 异常/分支预测失败时回滚
 *
 * 【为什么ROB是环形队列？】
 *   指令按程序序进入ROB（尾部），也按程序序从ROB提交出去（头部）。
 *   完美的先进先出 → 完美的环形队列应用场景！
 *
 * 【与简单队列的区别】
 *   ROB不是简单的FIFO：
 *   - 中间的项可以被随机修改（标记"执行完成"）
 *   - 需要按robIdx随机读/写某条指令的状态
 *   - 提交时需要从头部连续读出多条已完成的指令
 *
 * 所以这里我们不直接用CircularQueue模块，
 * 而是用 CircularQueuePtr 来管理指针，自己控制存储。
 * 这才是最灵活的用法！
 */
 
// ---- 第一步：定义ROB指针类型 ----
 
/**
 * ROB指针——继承CircularQueuePtr
 *
 * 这就是"旗子指针"的最直接使用：
 *   class RobPtr extends CircularQueuePtr[RobPtr](64)
 * 一行代码，就获得了 value+flag 的指针，
 * 以及前进、比较、判空满等全部功能
 */
class RobPtr(val entries: Int = 64) extends CircularQueuePtr[RobPtr](entries) {
  // 不需要额外添加任何字段！
  // CircularQueuePtr 已经提供了 value 和 flag
  // 您只需要继承它，就能获得完整的环形队列指针功能
}
 
// ---- 第二步：定义ROB中存储的数据 ----
 
/** ROB中每条指令的信息 */
class RobEntry extends Bundle {
  /** 指令是否已经执行完成 */
  val finished  = Bool()
  /** 指令是否有异常 */
  val hasException = Bool()
  /** 异常码 */
  val exceptionCode = UInt(8.W)
  /** 写回的目标寄存器编号 */
  val dstReg   = UInt(5.W)
  /** 写回的数据 */
  val dstData  = UInt(64.W)
  /** 是否需要写回寄存器 */
  val needWriteback = Bool()
}
 
// ---- 第三步：ROB模块 ----
 
class ROB(
  val robSize:    Int = 64,    // ROB项数
  val commitWidth: Int = 4,    // 每周期最多提交几条指令
  val dispatchWidth: Int = 4   // 每周期最多派发几条指令进入ROB
) extends Module {
  val io = IO(new Bundle {
    // ---- 派发端口：指令进入ROB ----
    val dispatch = Vec(dispatchWidth, Flipped(DecoupledIO(new Bundle {
      val entry    = new RobEntry   // 指令信息
      val robIdx   = Output(new RobPtr(robSize))  // 分配的ROB索引（返回给调度器）
    })))
 
    // ---- 写回端口：执行单元报告指令完成 ----
    val writeback = Vec(commitWidth * 2, Input(new Bundle {
      val robIdx  = new RobPtr(robSize)  // 哪条指令完成了
      val entry   = new RobEntry          // 完成后的信息
    }))
 
    // ---- 提交端口：按序提交已完成的指令 ----
    val commit = Vec(commitWidth, DecoupledIO(new Bundle {
      val entry   = new RobEntry
      val robIdx  = new RobPtr(robSize)
    }))
 
    // ---- 状态 ----
    val empty = Output(Bool())
    val full  = Output(Bool())
    val count = Output(UInt(log2Ceil(robSize + 1).W))
  })
 
  // =====================================================================
  // ROB指针管理
  // =====================================================================
 
  /**
   * headPtr: 提交指针，指向最老的未提交指令
   * tailPtr: 派发指针，指向下一个空槽位
   *
   * 这两个指针完全由 CircularQueuePtr 管理：
   * - 它们自带 value 和 flag
   * - 空满判断不需要额外逻辑
   * - 顺序比较天然支持
   */
  val headPtr = RegInit(0.U.asTypeOf(new RobPtr(robSize)))
  val tailPtr = RegInit(0.U.asTypeOf(new RobPtr(robSize)))
 
  // 空满判断 —— 旗子方案的精髓
  val empty = headPtr === tailPtr
  val full  = (headPtr.value === tailPtr.value) && (headPtr.flag =/= tailPtr.flag)
  val count = tailPtr.distanceTo(headPtr)
 
  io.empty := empty
  io.full  := full
  io.count := count
 
  // =====================================================================
  // 存储体
  // =====================================================================
 
  val data = Reg(Vec(robSize, new RobEntry))
 
  // =====================================================================
  // 派发逻辑：指令进入ROB
  // =====================================================================
 
  /**
   * 派发（入队）：
   *   第i条指令分配 tailPtr+i 的位置
   *   派发成功后 tailPtr 前进
   */
  val dispatchFireCnt = WireInit(0.U(log2Ceil(dispatchWidth + 1).W))
 
  for (i <- 0 until dispatchWidth) {
    val canDispatch = (count +& i.U) < robSize.U
    io.dispatch(i).ready := canDispatch && !full
 
    // 告诉调度器分配的ROB索引
    io.dispatch(i).bits.robIdx := {
      val ptr = Wire(new RobPtr(robSize))
      ptr.value := (tailPtr.value + i.U)(log2Ceil(robSize) - 1, 0)
      ptr.flag  := Mux(
        tailPtr.value + i.U >= robSize.U,
        !tailPtr.flag,
        tailPtr.flag
      )
      ptr
    }
 
    when (io.dispatch(i).fire) {
      val idx = (tailPtr.value + i.U)(log2Ceil(robSize) - 1, 0)
      data(idx) := io.dispatch(i).bits.entry
      // 新进入的指令还没执行完
      data(idx).finished := false.B
    }
  }
 
  dispatchFireCnt := PopCount(io.dispatch.map(_.fire))
  tailPtr := tailPtr + dispatchFireCnt
 
  // =====================================================================
  // 写回逻辑：执行单元报告完成（随机写！）
  // =====================================================================
 
  /**
   * 写回是ROB区别于简单FIFO的关键：
   *   指令可能乱序完成，需要按robIdx写入对应位置
   *   这不移动任何指针，只是修改存储体中的数据
   */
  for (wb <- io.writeback) {
    // 用robIdx.value 作为数组索引，直接写入
    // 这就是为什么索引必须是2的幂——位截断就是取余
    data(wb.robIdx.value).finished  := true.B
    data(wb.robIdx.value).dstData   := wb.entry.dstData
    data(wb.robIdx.value).hasException := wb.entry.hasException
    data(wb.robIdx.value).exceptionCode := wb.entry.exceptionCode
  }
 
  // =====================================================================
  // 提交逻辑：按序提交头部已完成的指令
  // =====================================================================
 
  /**
   * 提交（出队）：
   *   从headPtr开始，检查连续的已完成的指令
   *   遇到第一条未完成的就停止（必须按序！）
   *   提交成功后 headPtr 前进
   */
  val commitFireCnt = WireInit(0.U(log2Ceil(commitWidth + 1).W))
 
  // 计算每一条候选提交指令是否可以提交
  val canCommit = Wire(Vec(commitWidth, Bool()))
  for (i <- 0 until commitWidth) {
    val idx = (headPtr.value + i.U)(log2Ceil(robSize) - 1, 0)
    // 可以提交的条件：在ROB范围内 且 已执行完成 且 没有前面的指令阻挡
    val inRange = (count > i.U)
    val isFinished = data(idx).finished
    // 前面所有候选都已完成才能提交第i条（按序！）
    val noBlock = if (i == 0) true.B else canCommit(i - 1)
 
    canCommit(i) := inRange && isFinished && noBlock
 
    io.commit(i).valid := canCommit(i)
    io.commit(i).bits.entry  := data(idx)
    io.commit(i).bits.robIdx := {
      val ptr = Wire(new RobPtr(robSize))
      ptr.value := idx
      ptr.flag  := Mux(
        headPtr.value + i.U >= robSize.U,
        !headPtr.flag,
        headPtr.flag
      )
      ptr
    }
  }
 
  commitFireCnt := PopCount(io.commit.map(_.fire))
  headPtr := headPtr + commitFireCnt
}
